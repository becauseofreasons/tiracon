cmake_minimum_required(VERSION 3.25.0)

# Tiracon - Tiracon 6V emulation as VST3 plugin (Steinberg VST3 SDK, no JUCE)
#   Linux:   cmake -S vst3 -B ~/build/tiracon -G Ninja -DCMAKE_BUILD_TYPE=Release
#   Windows: cmake -S vst3 -B ~/build/tiracon-win -G Ninja -DCMAKE_BUILD_TYPE=Release
#                  -DCMAKE_TOOLCHAIN_FILE=vst3/cmake/mingw-w64.cmake
#   Win32:   cmake -S vst3 -B ~/build/tiracon-win32 -G Ninja -DCMAKE_BUILD_TYPE=Release
#                  -DCMAKE_TOOLCHAIN_FILE=vst3/cmake/mingw-w64-i686.cmake

option(TIRACON_BUILD_PLUGIN "Build the VST3 plugin (needs the VST3 SDK)" ON)
set(VST3_SDK_ROOT "$ENV{HOME}/SDKs/vst3sdk" CACHE PATH "Steinberg VST3 SDK folder")
set(TIRACON_ROM "${CMAKE_CURRENT_SOURCE_DIR}/firmware/tiracon_mod.bin" CACHE FILEPATH "Firmware built into the plugin")

project(Tiracon VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

#------------------------------------------------------------------ firmware ROM -> C++ array
if(NOT EXISTS "${TIRACON_ROM}")
    message(FATAL_ERROR "ROM not found: ${TIRACON_ROM}")
endif()
file(SIZE "${TIRACON_ROM}" _rom_size)
if(NOT _rom_size EQUAL 8192)
    message(FATAL_ERROR "ROM must be 8192 bytes: ${TIRACON_ROM} has ${_rom_size}")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${TIRACON_ROM}")
file(READ "${TIRACON_ROM}" _rom_hex HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _rom_c "${_rom_hex}")
string(REGEX REPLACE "(0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,0x..,)" "\\1\n  " _rom_c "${_rom_c}")
file(SHA256 "${TIRACON_ROM}" _rom_sha)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rom_data.cpp.in"
    "// generated by CMake from ${TIRACON_ROM}\n// SHA-256 ${_rom_sha}\n#include \"firmware.h\"\nnamespace t6v {\nconst unsigned char FIRMWARE_ROM[8192] = {\n  ${_rom_c}\n};\n}\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/rom_data.cpp.in" "${CMAKE_CURRENT_BINARY_DIR}/rom_data.cpp" COPYONLY)

#------------------------------------------------------------------ emulation core (no SDK, no JS)
add_library(tiracon_core STATIC
    src/core/z80.cpp
    src/core/devices.cpp
    src/core/board.cpp
    src/core/synth.cpp
    src/core/device.cpp
    src/core/firmware.cpp
    src/core/engine.cpp
    src/core/uimodel.cpp
    src/core/multiengine.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/rom_data.cpp"
)
target_include_directories(tiracon_core PUBLIC src/core)
# same floating point results as the WebApp: no fused multiply-add, no fast-math
target_compile_options(tiracon_core PUBLIC $<$<CXX_COMPILER_ID:GNU,Clang>:-ffp-contract=off>)

#------------------------------------------------------------------ comparison with the WebApp
add_executable(tiracon_compare test/compare_main.cpp)
target_link_libraries(tiracon_compare PRIVATE tiracon_core)
if(MINGW)
    target_link_options(tiracon_compare PRIVATE -static)
endif()

add_executable(tiracon_multitest test/multitest.cpp)
target_link_libraries(tiracon_multitest PRIVATE tiracon_core)
if(MINGW)
    target_link_options(tiracon_multitest PRIVATE -static)
endif()

#------------------------------------------------------------------ VST3 plugin
if(TIRACON_BUILD_PLUGIN)
    include(cmake/plugin.cmake)
endif()
